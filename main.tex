%% Abstract (structured: problem, gap, approach, result, claim)
%% ------------------------------------------------------------------
\begin{abstract}
\todo{Problem: volumetric video over private \fiveg to untethered
\glspl{hmd}; production \quic is reliable, so loss never reaches the decoder
-- it reaches the clock.}
\todo{Gap: existing network-aware QoE models for volumetric media
(Udora et al.) fit five coefficients per quality level on reliable
transport and generalise poorly (held-out PLCC 0.57--0.66); no model
explains \emph{why} delay and loss degrade QoE.}
\todo{Approach: \crystal, a two-layer model. Layer~1 derives the temporal
loss fraction $\tlf$ from transport parameters (slack, RTT, recovery
cost, \gof structure) with no perceptual free parameter; Layer~2 maps
$\tlf$ to MOS with one parameter, $\crystalmodel$.}
\todo{Result: ITU-T P.910 ACR study ($N=50$), $2\times2$ factorial over
loss and latency; separability refuted ($1.61\times$ interaction);
\gof size, not codec identity, governs resilience; single $\alpha$
predicts held-out conditions.}
\todo{Claim: under reliable transport, latency and loss act exclusively
through $\tlf$, mediated by the codec's delivery-unit structure.}
\end{abstract}
